cmake_minimum_required(VERSION 3.23)
project(mk8dx-quickscore VERSION 0.8.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
if(MSVC)
  add_compile_options(/utf-8)
endif()
set(QUICKSCORE_DEPENDENCIES "" CACHE PATH "Prepared OBS SDK directory containing include/ and lib/")
set(CPPWINRT_GENERATED "" CACHE PATH "Optional generated C++/WinRT projection")
if(QUICKSCORE_DEPENDENCIES)
  foreach(component IN ITEMS libobs obs-frontend-api)
    if(component STREQUAL "libobs")
      set(library_name obs)
    else()
      set(library_name obs-frontend-api)
    endif()
    add_library(OBS::${component} UNKNOWN IMPORTED)
    set_target_properties(OBS::${component} PROPERTIES
      IMPORTED_LOCATION "${QUICKSCORE_DEPENDENCIES}/lib/${library_name}.lib"
      INTERFACE_INCLUDE_DIRECTORIES "${QUICKSCORE_DEPENDENCIES}/include")
  endforeach()
else()
  find_package(libobs REQUIRED)
  find_package(obs-frontend-api REQUIRED)
endif()
find_package(Qt6 REQUIRED COMPONENTS Core Gui Widgets Network)

set(core_sources src/core/analysis.cpp src/core/finish-notice.cpp src/core/native-ocr.cpp
    src/core/scoring.cpp src/core/session-store.cpp src/reference/image.cpp src/reference/recognizer.cpp
    src/core/fast-reader.cpp)
set(dock_sources src/ui/main-dock.cpp src/obs/frame-capture.cpp src/obs/scoreboard-source.cpp
    src/core/credential-store.cpp src/core/discord-sender.cpp src/core/local-reader.cpp
    src/core/result-capture.cpp src/core/jev-api-reader.cpp)
function(configure_core target)
  target_include_directories(${target} PRIVATE src/core src/obs src/ui)
  if(CPPWINRT_GENERATED)
    target_include_directories(${target} BEFORE PRIVATE "${CPPWINRT_GENERATED}")
  endif()
  target_link_libraries(${target} PRIVATE Qt6::Core Qt6::Gui)
  if(WIN32)
    target_compile_definitions(${target} PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
    target_link_libraries(${target} PRIVATE windowsapp windowscodecs ole32 advapi32)
  endif()
  if(MSVC)
    target_compile_options(${target} PRIVATE /await /utf-8 /W3)
  endif()
endfunction()
add_library(mk8dx-quickscore MODULE src/plugin-main.cpp ${core_sources} ${dock_sources})
configure_core(mk8dx-quickscore)
target_link_libraries(mk8dx-quickscore PRIVATE OBS::libobs OBS::obs-frontend-api Qt6::Widgets Qt6::Network)
install(TARGETS mk8dx-quickscore LIBRARY DESTINATION bin/64bit RUNTIME DESTINATION bin/64bit)
install(DIRECTORY data/ DESTINATION data PATTERN "__pycache__" EXCLUDE PATTERN "*.pyc" EXCLUDE)
get_target_property(qt_core_location Qt6::Core IMPORTED_LOCATION_RELEASE)
get_filename_component(qt_bin "${qt_core_location}" DIRECTORY)
get_filename_component(qt_root "${qt_bin}" DIRECTORY)
if(WIN32 AND EXISTS "${qt_root}/plugins/tls/qschannelbackend.dll")
  set(QUICKSCORE_TLS_BACKEND "${qt_root}/plugins/tls/qschannelbackend.dll" CACHE FILEPATH "TLS backend matching recipient OBS Qt runtime")
  install(FILES "${QUICKSCORE_TLS_BACKEND}" DESTINATION bin/64bit/tls)
endif()
